double totalCompra = 0.0;
char continuarComprando = "s";

Console.WriteLine("--- Carrinho de Compras ---");

while (continuarComprando == "s"){

    Console.Write("Digite o preço do item: ");
    double preco = double.Parse(Console.ReadLine());

    totalCompra += preco;

    Console.Write("Deseja adicionar mais itens? (s/n):");
    continuarComprando = Console.ReadLine();
}

Console.WriteLine($"Total da sua compra: R$ {totalCompra:F2}");
